   function getResolutionValue(str) {
      const width = window.innerWidth;
      const ranges = str
         .split(",")
         .map((pair) => {
            const [breakpoint, value] = pair.split(":");
            return { breakpoint: Number(breakpoint), value: Number(value) };
         })
         .sort((a, b) => a.breakpoint - b.breakpoint);
      let lastValue = ranges[0].value;
      for (const { breakpoint, value } of ranges) {
         if (width < breakpoint) {
            return lastValue;
         }
         lastValue = value;
      }
      return lastValue;
   }

      filtdis = prfilters[1] ? prfilters[1].getAttribute("data-elem-id") : "";
      if (startact) {
         filtact = startact.getAttribute("data-elem-id");
         startact.setAttribute("data-elem-id", filtdis);
         startact.className = startact.className.replace(
            "vfilteract",
            "vfilter",
         );
      }
      if (prfilters.length > 0) {
         prfilters.forEach(function (element) {
            if (element.getAttribute("data-elem-type") !== "form")
               element.addEventListener("click", function (e) {
                  prclear.forEach(function (clearing) {
                     clearing.style.display = "block";
                  });
                  vrefreshcat(
                     sectn,
                     cardstotal,
                     e.target.getAttribute("href").replace("#", "") ||
                        e.target
                           .closest("[href]")
                           .getAttribute("href")
                           .replace("#", ""),
                  );
               });
            else
               element.addEventListener("change", function () {
                  curpage = 1;
                  prclear.forEach(function (clearing) {
                     clearing.style.display = "block";
                  });
                  let sl = this.querySelector("select");
                  let vl = Array.from(
                     sl.querySelectorAll('[value]:not([value=""])'),
                  ).map((v) => v.value.trim());
                  vrefreshcat(sectn, cardstotal, {
                     value: sl.value.trim(),
                     groupValues: vl,
                  });
               });
         });
      }
      setTimeout(() => {
         document
            .querySelectorAll("." + mainclass + " .vsearch .tn-form__submit")
            .forEach(function (sbtorem) {
               sbtorem.remove();
            });
         document
            .querySelectorAll(
               "." + mainclass + ' [class*="vfilter"] .tn-form__submit',
            )
            .forEach(function (sbtorem) {
               sbtorem.remove();
            });
         document
            .querySelectorAll("." + mainclass + " .vsort .tn-form__submit")
            .forEach(function (sbtorem) {
               sbtorem.remove();
            });
         document
            .querySelectorAll(
               "." + mainclass + ' [class*="vpricem"] .tn-form__submit',
            )
            .forEach(function (sbtorem) {
               sbtorem.remove();
            });
      }, 500);
      let srt = document.querySelector("." + mainclass + " .vsort");
      if (srt) {
         srt.addEventListener("change", function (e) {
            sort = e.target.selectedIndex;
            vrefreshcat(sectn, cardstotal, false);
         });
      }
      let prmax = document.querySelector("." + mainclass + " .vpricemax");
      if (prmax) {
         prmax.addEventListener("change", function (e) {
            if (e.target.value) {
               prclear.forEach(function (clearing) {
                  clearing.style.display = "block";
               });
               pmax = e.target.value.replaceAll(" ", "");
            }
            vrefreshcat(sectn, cardstotal, false);
         });
         let prmin = document.querySelector("." + mainclass + " .vpricemin");
         if (prmin) {
            prmin.addEventListener("change", function (e) {
               if (e.target.value) {
                  prclear.forEach(function (clearing) {
                     clearing.style.display = "block";
                  });
                  pmin = e.target.value.replaceAll(" ", "");
               }
               vrefreshcat(sectn, cardstotal, false);
            });
         }
      }
      document
         .querySelectorAll("." + mainclass + ' [class*="vpricem"] input')
         .forEach(function (oldp) {
            oldp.addEventListener("input", function (e) {
               vformatNumber(e.target);
            });
         });
      let vsearchbtn = document.querySelector("." + mainclass + " .vsearchbtn");
      if (vsearchbtn) {
         vsearchbtn.addEventListener("click", () => {
            prclear.forEach(function (clearing) {
               clearing.style.display = "block";
            });
            searchedw = document.querySelector(
               "." + mainclass + " .vsearch .t-input",
            ).value;
            vrefreshcat(sectn, cardstotal, false);
         });
      }
      let vsearchInput = document.querySelector(
         "." + mainclass + " .vsearch .t-input",
      );
      if (vsearchInput) {
         vsearchInput.addEventListener("keydown", function (event) {
            if (event.key === "Enter") {
               event.preventDefault();
               prclear.forEach(function (clearing) {
                  clearing.style.display = "block";
               });
               searchedw = vsearchInput.value;
               vrefreshcat(sectn, cardstotal, false);
            }
         });
      }
      isrefresh = false;
      if (window.location.search) {
         window.location.search
            .replace("?", "")
            .split("&")
            .forEach((item) => {
               const [key, value] = item.split("=");
               linksearch[key] = isNaN(value) ? value : Number(value);
               if (linksearch.vcatsection) {
                  let targ = document.querySelector(
                     "." +
                        mainclass +
                        ' .vcatsection [href="#' +
                        decodeURIComponent(linksearch.vcatsection) +
                        '"]',
                  );
                  targ.click();
                  document
                     .querySelector(".vcatsectionact")
                     .classList.remove("vcatsectionact");
                  targ.classList.add("vcatsectionact");
               }
               if (linksearch.vfilter)
                  document
                     .querySelector(
                        "." +
                           mainclass +
                           ' .vfilter [href="#' +
                           decodeURIComponent(linksearch.vfilter) +
                           '"]',
                     )
                     .click();
               if (linksearch.vpricemax) {
                  document.querySelector(
                     "." + mainclass + " .vpricemax input",
                  ).value = linksearch.vpricemax;
                  pmax = linksearch.vpricemax;
               }
               if (linksearch.vpricemin) {
                  document.querySelector(
                     "." + mainclass + " .vpricemin input",
                  ).value = linksearch.vpricemin;
                  pmin = linksearch.vpricemin;
               }
               prclear.forEach(function (clearing) {
                  clearing.style.display = "block";
               });
               if (linksearch.vsearch) {
                  searchedw = linksearch.vsearch;
                  document.querySelector(
                     "." + mainclass + " .vsearch .t-input",
                  ).value = decodeURIComponent(linksearch.vsearch);
               }
            });
         isrefresh = true;
         await vrefreshcat(linksearch.vcatsection, cardstotal);
      } else {
         isrefresh = true;
         await vrefreshcat(`${sectn ? sectn : ""}`, cardstotal);
      }
      document
         .querySelectorAll("." + mainclass + ` [class*="vcatbuy"]`)
         .forEach(function (element) {
            element.addEventListener("click", function (e) {
               let quant = document.querySelector(
                  "." +
                     mainclass +
                     " .vcatquant" +
                     (getnum(element.className, "vcatbuy") + 1) +
                     " input",
               );
               if (quant) quant.value = 1;
            });
         });
      document
         .querySelectorAll("." + mainclass + ' [class*="vcatquant"]')
         .forEach(function (element) {
            if (element) {
               let submitButton = element.querySelector(".tn-form__submit");
               if (submitButton) {
                  submitButton.remove();
               }
               element.addEventListener("input", function (e) {
                  let buyElement = document.querySelector(
                     "." +
                        mainclass +
                        " .vcatbuy" +
                        (getnum(element.className, "vcatquant") + 1),
                  );
                  if (e.target.value < 1) e.target.value = 1;
                  else if (
                     buyElement &&
                     parseFloat(e.target.value) >
                        parseFloat(buyElement.getAttribute("data-product-inv"))
                  ) {
                     let maxInput = buyElement.querySelector(
                        ".t-store__prod__quantity input",
                     );
                     if (maxInput) {
                        e.target.value = maxInput.max;
                     }
                  } else if (buyElement) {
                     let quantityInput = buyElement.querySelector(
                        ".t-store__prod__quantity input",
                     );
                     if (quantityInput) {
                        quantityInput.value = e.target.value || 1;
                     }
                  }
               });
            }
         });
      document
         .querySelector("." + mainclass)
         .addEventListener("click", function (e) {
            if (
               e.target &&
               e.target.hash === "" &&
               document.querySelector(
                  '[data-tooltip-hook=""] form [name="tovar"]',
               )
            )
               document.querySelector(
                  '[data-tooltip-hook=""] form [name="tovar"]',
               ).value =
                  e.target.getAttribute("data-name") ||
                  e.target.closest("[href]").getAttribute("data-name");
         });
      document
         .querySelectorAll("." + mainclass + ' [class*="vcatvar"]')
         .forEach(function (element) {
            element.addEventListener("change", function (e) {
               let cl = getnum(element.className, "vcatvar") + 1;
               let id = parseFloat(
                  document
                     .querySelector("." + mainclass + " .vcatbuy" + cl)
                     .getAttribute("data-product-uid"),
               );
               let trg = document.querySelector(
                  "." +
                     mainclass +
                     " .vcatbuy" +
                     cl +
                     ' select [value="' +
                     e.target.value +
                     '"]',
               );
               if (trg) {
                  trg.selected = true;
                  voronUpdateOne(
                     mainclass,
                     voroncatproducts[mainclass].products.find((item) =>
                        item.editions.some((edition) => edition.uid === id),
                     ),
                     cl,
                  );
               } else {
                  window.alert(
                     "Товара с такими характеристиками нет в наличии. Пожалуйста, выберите другие характеристики.",
                  );
               }
            });
         });
   };
   window.addEventListener(
      "load",
      function () {
         voroncatinit(
            "uc-catalog",
            getResolutionValue(""),
            "uc-paginationBlock",
            160999850272,
         );
      },
      { once: true },
   );
</script>
